using Verse;

namespace RimWorld;

public class TileMutatorWorker_CaveLakes : TileMutatorWorker_Caves
{
	private const float LakeChancePerCell = 0.005f;

	private static readonly IntRange DeepWaterSizeRange = new IntRange(25, 45);

	private static readonly IntRange ShallowWaterSizeRange = new IntRange(110, 150);

	public TileMutatorWorker_CaveLakes(TileMutatorDef def)
		: base(def)
	{
	}

	public override void GeneratePostTerrain(Map map)
	{
		if (!ModsConfig.OdysseyActive)
		{
			return;
		}
		MapGenFloatGrid caves = MapGenerator.Caves;
		base.GeneratePostTerrain(map);
		foreach (IntVec3 allCell in map.AllCells)
		{
			if (caves[allCell] <= 0f || allCell.GetTerrain(map).IsWater || !Rand.Chance(0.005f))
			{
				continue;
			}
			foreach (IntVec3 item in GridShapeMaker.IrregularLump(allCell, map, ShallowWaterSizeRange.RandomInRange + 1))
			{
				item.GetEdifice(map)?.Destroy();
				map.terrainGrid.SetTerrain(item, MapGenUtility.ShallowFreshWaterTerrainAt(item, map));
			}
			foreach (IntVec3 item2 in GridShapeMaker.IrregularLump(allCell, map, DeepWaterSizeRange.RandomInRange))
			{
				map.terrainGrid.SetTerrain(item2, MapGenUtility.DeepFreshWaterTerrainAt(item2, map));
			}
		}
	}
}
