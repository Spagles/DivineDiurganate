using System.Collections.Generic;
using System.Linq;
using RimWorld.SketchGen;
using Verse;

namespace RimWorld;

public class GenStep_OrbitalRelay : GenStep
{
	private OrbitalDebrisDef orbitalDebrisDef;

	private static readonly IntRange NumProcessorArrays = new IntRange(2, 3);

	public override int SeedPart => 982398321;

	public override void Generate(Map map, GenStepParams parms)
	{
		if (!ModLister.CheckOdyssey("Orbital Relay"))
		{
			return;
		}
		map.regionAndRoomUpdater.Enabled = true;
		CellRect rect = CellRect.CenteredOn(map.Center, 25);
		GeneratePlatform(map, rect);
		List<CellRect> usedRects = MapGenerator.GetOrGenerateVar<List<CellRect>>("UsedRects");
		SketchResolveParams sketchResolveParams = default(SketchResolveParams);
		sketchResolveParams.sketch = new Sketch();
		sketchResolveParams.rect = new CellRect(0, 0, 9, 9);
		SketchResolveParams parms2 = sketchResolveParams;
		Sketch sketch = RimWorld.SketchGen.SketchGen.Generate(SketchResolverDefOf.AncientConcretePad, parms2);
		sketch.Spawn(map, map.Center, null, Sketch.SpawnPosType.OccupiedCenter, Sketch.SpawnMode.Normal, wipeIfCollides: true, forceTerrainAffordance: true, clearEdificeWhereFloor: true);
		usedRects.Add(parms2.rect.Value.MovedBy(map.Center - sketch.OccupiedCenter));
		Thing thing = ThingMaker.MakeThing(ThingDefOf.MechRelay);
		GenPlace.TryPlaceThing(thing, map.Center, map, ThingPlaceMode.Near);
		thing.TryGetComp<CompMechRelay>().inert = true;
		rect.TryFindRandomInnerRect(new IntVec2(12, 12), out var rect2, (CellRect r) => !usedRects.Any((CellRect ur) => ur.Overlaps(r)) && r.Cells.All((IntVec3 c) => c.GetTerrain(map) != TerrainDefOf.Space));
		usedRects.Add(rect2);
		for (int i = 0; i < NumProcessorArrays.RandomInRange; i++)
		{
			int width = Rand.RangeInclusive(1, 2) * (ThingDefOf.MechanoidProcessor.size.x + 1) + 1;
			int height = Rand.RangeInclusive(1, 2) * (ThingDefOf.MechanoidProcessor.size.z + 1) + 1;
			if (TryGetFreeRect(map, width, height, usedRects, out var rect3))
			{
				GenerateSketch(SketchResolverDefOf.MechhiveProcessorArray, map, rect3);
				usedRects.Add(rect3);
			}
		}
		MapGenerator.PlayerStartSpot = rect2.CenterCell;
		map.OrbitalDebris = orbitalDebrisDef;
	}

	private void GeneratePlatform(Map map, CellRect rect)
	{
		foreach (IntVec3 cell in rect.Cells)
		{
			if (cell.InHorDistOf(map.Center, (float)rect.Width / 2f))
			{
				map.terrainGrid.SetTerrain(cell, TerrainDefOf.MechanoidPlatform);
			}
		}
	}

	private void GenerateSketch(SketchResolverDef sketchDef, Map map, CellRect rect)
	{
		SketchResolveParams parms = default(SketchResolveParams);
		parms.sketch = new Sketch();
		parms.rect = new CellRect(0, 0, rect.Width, rect.Height);
		RimWorld.SketchGen.SketchGen.Generate(sketchDef, parms).Spawn(map, rect.CenterCell, Faction.OfMechanoids, Sketch.SpawnPosType.OccupiedCenter, Sketch.SpawnMode.Normal, wipeIfCollides: true, forceTerrainAffordance: true, clearEdificeWhereFloor: true);
	}

	private bool TryGetFreeRect(Map map, int width, int height, List<CellRect> usedRects, out CellRect rect)
	{
		rect = CellRect.Empty;
		if (!CellFinder.TryFindRandomCell(map, delegate(IntVec3 c)
		{
			CellRect tmpRect = CellRect.CenteredOn(c, width, height);
			return tmpRect.InBounds(map) && usedRects.All((CellRect ur) => !ur.Overlaps(tmpRect)) && tmpRect.Cells.All((IntVec3 cell) => cell.GetTerrain(map) == TerrainDefOf.MechanoidPlatform);
		}, out var result))
		{
			return false;
		}
		rect = CellRect.CenteredOn(result, width, height);
		return true;
	}
}
